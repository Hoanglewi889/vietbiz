[요소 설명] 프로젝트가 없을 때는 빈 폴더가 천천히 떠 있고, 템플릿에 손을 올리면 폴더가 살짝 벌어지며 종이가 고개를 내밉니다. 템플릿을 고르면 종이가 날아 들어가 폴더가 꿀꺽 삼킨 뒤 첫 항목이 목록으로 바뀌고, 모두 지우면 폴더가 다시 내려앉습니다.
[조작] 템플릿 버튼에 마우스를 올렸다가 눌러 첫 프로젝트를 만들어 보세요. 목록의 휴지통으로 모두 지우면 빈 화면이 돌아옵니다

    b.addEventListener('click', function () { create(k, b); });
    tplBox.appendChild(b);
  });
  var tpls = tplBox.querySelectorAll('button');
  var made = [0, 0, 0], count = 0, busy = false, rot = 0, ts = [];
  function later(fn, ms) { ts.push(setTimeout(function () { if (root.isConnected) fn(); }, ms)); }
  function setCount(v) {
    var dir = v > count ? 1 : -1;
    count = v;
    n.textContent = v;
    if (!rm) n.animate([{ transform: 'translateY(' + dir * 8 + 'px)', opacity: 0 }, { transform: 'none', opacity: 1 }], { duration: 320, easing: 'cubic-bezier(.2,.8,.2,1)' });
    var full = v >= MAX;
    add.setAttribute('aria-disabled', String(full));
    more.setAttribute('aria-disabled', String(full));
    more.lastChild.textContent = full ? '최대 5개까지 만들 수 있어요' : '새 프로젝트';
  }
  function insert(k) {
    made[k]++;
    var t = T[k], name = t[1] + (made[k] > 1 ? ' ' + made[k] : '');
    var li = document.createElement('li');
    li.className = 'u088-it new';
    li.style.setProperty('--c', t[2]);
    li.innerHTML = '<div class="u088-clip"><div class="u088-row"><span class="u088-ic" aria-hidden="true"><svg viewBox="0 0 16 16">' + t[3] + '</svg></span><span class="u088-tx"><b></b><small></small></span><button class="u088-del" type="button"><svg viewBox="0 0 16 16" aria-hidden="true"><path d="M3 4.2h10M6.2 4.2V2.6h3.6v1.6M4.6 4.2l.6 9.2h5.6l.6-9.2"/></svg></button></div></div>';
    li.querySelector('b').textContent = name;
    li.querySelector('small').textContent = '방금 만듦 · ' + t[0];
    li.querySelector('.u088-del').setAttribute('aria-label', name + ' 삭제');
    list.insertBefore(li, list.firstChild);
    var lost = root.querySelector('.u088-empty').contains(document.activeElement);
    root.classList.add('has');
    setCount(count + 1);
    live.textContent = '‘' + name + '’ 프로젝트를 만들었어요';
    if (lost) more.focus({ preventScroll: true });
    later(function () { li.classList.remove('new'); }, 1400);
  }
  // 첫 프로젝트일 때만 종이가 버튼에서 폴더로 포물선을 그리며 날아가고, 폴더가 삼킨 뒤 목록으로 바뀝니다
  function create(k, from) {
    if (busy || count >= MAX) return;
    if (count > 0 || rm) { insert(k); return; }
    busy = true;
    var rr = root.getBoundingClientRect(), s = root.offsetWidth / rr.width;
    var fr = from.getBoundingClientRect(), ar = art.getBoundingClientRect();
    var x0 = (fr.left + fr.width / 2 - rr.left) * s, y0 = (fr.top + fr.height / 2 - rr.top) * s;
    var x1 = (ar.left + ar.width / 2 - rr.left) * s, y1 = (ar.top + ar.height * 0.3 - rr.top) * s;
    var dx = x1 - x0, dy = y1 - y0;
    fly.style.setProperty('--c', T[k][2]);
    fly.style.left = (x0 - 13) + 'px'; fly.style.top = (y0 - 16) + 'px';
    art.classList.add('eat');
    fly.animate([
      { transform: 'translate(0,0) rotate(-10deg) scale(.5)', opacity: 0 },
      { transform: 'translate(' + dx * 0.45 + 'px,' + (dy * 0.45 - 70) + 'px) rotate(12deg) scale(1.1)', opacity: 1, offset: 0.45 },
      { transform: 'translate(' + dx + 'px,' + dy + 'px) rotate(0deg) scale(.8)', opacity: 1, offset: 0.82 },
      { transform: 'translate(' + dx + 'px,' + (dy + 26) + 'px) rotate(0deg) scale(.6)', opacity: 0 }
    ], { duration: 760, easing: 'cubic-bezier(.45,0,.25,1)' });
    later(function () { art.classList.remove('eat'); art.classList.add('gulp'); }, 700);
    later(function () { art.classList.remove('gulp'); insert(k); busy = false; }, 1180);
  }
  add.addEventListener('click', function () { create(rot++ % 3, add); });
  more.addEventListener('click', function () { create(rot++ % 3, more); });
  // 지운 줄은 높이가 접히며 사라지고, 마지막 하나를 지우면 폴더가 위에서 다시 내려앉습니다
  list.addEventListener('click', function (e) {
    var b = e.target.closest('.u088-del');
    if (!b) return;
    var li = b.closest('li');
    if (li.classList.contains('out')) return;
    var had = li.contains(document.activeElement);
    li.classList.add('out');
    setCount(count - 1);
    live.textContent = '‘' + li.querySelector('b').textContent + '’ 프로젝트를 지웠어요';
    if (count === 0) later(function () {
      if (count !== 0) return;
      root.classList.remove('has');
      art.classList.remove('drop'); void art.offsetWidth; art.classList.add('drop');
      later(function () { art.classList.remove('drop'); }, 900);
      if (had) tpls[0].focus({ preventScroll: true });
    }, rm ? 0 : 160);
    later(function () {
      li.remove();
      if (had && count > 0) {
        var left = list.querySelector('.u088-it:not(.out) .u088-del');
        (left || more).focus({ preventScroll: true });
      }
    }, rm ? 0 : 380);
  });
})();
</script>
