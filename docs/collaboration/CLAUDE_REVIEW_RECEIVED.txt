Review VietBiz Sprint07: quyền truy cập và tính nguyên tử (documents + invitations)

Kết luận nhanh
Invitations (accept, revoke, invite): không thấy lỗi nguyên tử. Các hàm này đã dùng UPDATE … WHERE status='pending' AND revoked_at IS NULL rồi kiểm tra rowcount. Partial unique index chặn trường hợp hai lời mời cùng pending. Mọi HTTPException bên trong with conn() đều làm transaction rollback, kể cả khi authorize(inviter) thất bại sau bước claim. Không nên sửa phần này.
Documents: có 1 lỗi nghiêm trọng (race khi review) và 2 lỗi nhỏ hơn, mô tả bên dưới.
Lỗi 1 — HIGH: race approve/reject trong review()

Vị trí: services/api/app/main.py, hàm review().

Bằng chứng: hàm đọc SELECT status rồi kiểm tra != 'needs_review' trong Python. Sau đó lệnh UPDATE documents SET status=? WHERE id=? chạy không có điều kiện status.

Trên SQLite (pysqlite qua SQLAlchemy): BEGIN chỉ được phát ra trước câu DML đầu tiên, nên SELECT không giữ lock. Hai request cùng đọc thấy needs_review. Cả hai cùng UPDATE và commit lần lượt.
Trên PostgreSQL READ COMMITTED: request B bị chặn ở UPDATE. Khi A commit, B đánh giá lại WHERE id=?, điều kiện vẫn đúng, nên B ghi đè kết quả của A.

Hậu quả:

Cả hai request đều nhận 200.
Trạng thái cuối là của request đến sau, ví dụ approve bị ghi đè thành reject.
Audit log có hai sự kiện mâu thuẫn nhau (document.approved và document.rejected).

Với chứng từ kế toán, đây là lỗi toàn vẹn dữ liệu. UPDATE cũng thiếu organization_id. Hiện tại chưa khai thác được vì SELECT đã lọc theo org, nhưng nên thêm để phòng thủ nhiều lớp.

# services/api/tests/test_document_review_atomicity.py
import threading
from contextlib import contextmanager
from fastapi.testclient import TestClient
from app import main

PDF = b'%PDF-1.4\n%%EOF\n'
REVIEW_EVENTS = {'document.approved', 'document.rejected', 'document.needs_changes'}


def _org_with_doc(client):
    owner = client.post('/dev/users', params={'display_name': 'Owner'}).json()['id']
    h = {'X-User': owner}
    org = client.post('/v1/organizations', json={'name': 'Race s.r.o.'}, headers=h).json()['id']
    r = client.post(f'/v1/organizations/{org}/documents', headers=h,
                    files={'file': ('a.pdf', PDF, 'application/pdf')})
    assert r.status_code == 201
    return h, org, r.json()['id']


def _run_concurrent_reviews(monkeypatch, h, org, doc, decisions):
    barrier = threading.Barrier(len(decisions), timeout=5)
    real_conn = main.conn

    class Gate:
        def __init__(self, tx):
            self.tx = tx

        def execute(self, sql, params=()):
            if sql.lstrip().upper().startswith('UPDATE DOCUMENTS'):
                barrier.wait()  # force both requests to reach the write together
            return self.tx.execute(sql, params)


    @contextmanager
    def gated():
        with real_conn() as tx:
            yield Gate(tx)

    monkeypatch.setattr(main, 'conn', gated)
    results = []

    def run(decision):
        with TestClient(main.app) as c:  # one client per thread
            results.append(c.post(f'/v1/organizations/{org}/documents/{doc}/review',
                                  json={'decision': decision}, headers=h))

    threads = [threading.Thread(target=run, args=(d,)) for d in decisions]
    for t in threads:
        t.start()
    for t in threads:
        t.join(15)
    monkeypatch.setattr(main, 'conn', real_conn)
    return results


def _assert_single_winner(client, h, org, results):
    assert sorted(r.status_code for r in results) == [200, 409]
    winner = next(r.json()['status'] for r in results if r.status_code == 200)
    docs = client.get(f'/v1/organizations/{org}/documents', headers=h).json()
    assert docs[0]['status'] == winner
    audit = client.get(f'/v1/organizations/{org}/audit', headers=h).json()
    assert [e['event'] for e in audit if e['event'] in REVIEW_EVENTS] == ['document.' + winner]



def test_concurrent_approve_and_reject_only_one_wins(client, monkeypatch):
    h, org, doc = _org_with_doc(client)
    results = _run_concurrent_reviews(monkeypatch, h, org, doc, ['approve', 'reject'])
    _assert_single_winner(client, h, org, results)


def test_two_identical_review_requests_only_one_succeeds(client, monkeypatch):
    h, org, doc = _org_with_doc(client)
    results = _run_concurrent_reviews(monkeypatch, h, org, doc, ['approve', 'approve'])
    _assert_single_winner(client, h, org, results)


def test_review_unknown_document_still_404(client):
    h, org, _ = _org_with_doc(client)
    r = client.post(f'/v1/organizations/{org}/documents/does-not-exist/review',
                    json={'decision': 'approve'}, headers=h)
    assert r.status_code == 404


def test_download_missing_original_is_not_audited(client):
    h, org, doc = _org_with_doc(client)
    for p in (main.FILES / org).iterdir():
        p.unlink()
    r = client.get(f'/v1/organizations/{org}/documents/{doc}/original', headers=h)
    assert r.status_code == 404  # old code: RuntimeError/500
    audit = client.get(f'/v1/organizations/{org}/audit', headers=h).json()
    assert not any(e['event'] == 'document.downloaded' for e in audit)
